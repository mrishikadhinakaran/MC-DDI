═══════════════════════════════════════════════════════════════════════════════

Step 3.1: Create Preprocessing Script
   In VS Code, create file: preprocess_data.py

   Copy-paste this EXACTLY:

   ═════════════════════════════════════════════════════════════════
   import pandas as pd
   import numpy as np
   from rdkit import Chem
   from rdkit.Chem import AllChem
   from tqdm import tqdm

   print("Loading data...")
   df = pd.read_csv('drugbank_ddi_data.csv')
   print(f"✓ Loaded {len(df)} interactions")

   # Get unique drugs
   unique_drugs = list(set(df['Drug1_ID'].unique()) | set(df['Drug2_ID'].unique()))
   print(f"✓ Total unique drugs: {len(unique_drugs)}")

   # Get SMILES for each drug using TDC internal data
   print("Fetching SMILES strings...")
   from tdc.multi_pred import DDI
   data = DDI(name='DrugBank')
   raw_data = data.get_data()

   # Create SMILES mapping
   smiles_dict = {}
   for idx, row in raw_data.iterrows():
       if pd.notna(row.get('Drug1')):
           drug_id = row['Drug1_ID']
           smiles = row.get('Drug1')
           if pd.notna(smiles):
               smiles_dict[drug_id] = smiles
       if pd.notna(row.get('Drug2')):
           drug_id = row['Drug2_ID']
           smiles = row.get('Drug2')
           if pd.notna(smiles):
               smiles_dict[drug_id] = smiles

   print(f"✓ Obtained SMILES for {len(smiles_dict)} drugs")

   # Convert SMILES to Morgan fingerprints
   print("Converting SMILES to fingerprints...")
   fp_dict = {}

   for drug_id in tqdm(unique_drugs):
       if drug_id not in smiles_dict:
           fp_dict[drug_id] = np.zeros(2048)
       else:
           try:
               smiles = smiles_dict[drug_id]
               mol = Chem.MolFromSmiles(smiles)
               if mol is not None:
                   fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, nBits=2048)
                   fp_dict[drug_id] = np.array(fp, dtype=np.float32)
               else:
                   fp_dict[drug_id] = np.zeros(2048)
           except:
               fp_dict[drug_id] = np.zeros(2048)

   print(f"✓ Generated fingerprints for {len(fp_dict)} drugs")

   # Create paired fingerprints for DDI
   print("Creating paired drug fingerprints...")
   X = []
   y = []

   for idx, row in tqdm(df.iterrows(), total=len(df)):
       drug1_id = row['Drug1_ID']
       drug2_id = row['Drug2_ID']

       if drug1_id in fp_dict and drug2_id in fp_dict:
           fp_pair = np.concatenate([fp_dict[drug1_id], fp_dict[drug2_id]])
           X.append(fp_pair)
           y.append(row['Y'])

   X = np.array(X)
   y = np.array(y)

   print(f"✓ Created {len(X)} paired fingerprints")
   print(f"✓ Feature shape: {X.shape}")
   print(f"✓ Label distribution: {np.unique(y, return_counts=True)}")

   # Save preprocessed data
   np.save('X_features.npy', X)
   np.save('y_labels.npy', y)
   print("✓ Saved features to X_features.npy")
   print("✓ Saved labels to y_labels.npy")
   ═════════════════════════════════════════════════════════════════
